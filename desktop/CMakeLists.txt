cmake_minimum_required(VERSION 3.22)
project(EndlessDesktop VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(EFFECT "junoChorus" CACHE STRING "Catalog effect ID")
file(READ "${REPO_ROOT}/desktop.lock.json" LOCK)
string(JSON JUCE_COMMIT GET "${LOCK}" juce commit)
string(JSON JUCE_LOCAL GET "${LOCK}" juce localSource)
set(JUCE_SOURCE_DIR "${REPO_ROOT}/${JUCE_LOCAL}" CACHE PATH "Pinned JUCE checkout")
execute_process(COMMAND git -C "${JUCE_SOURCE_DIR}" rev-parse HEAD OUTPUT_VARIABLE ACTUAL_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE GIT_RESULT)
execute_process(COMMAND git -C "${JUCE_SOURCE_DIR}" status --porcelain OUTPUT_VARIABLE JUCE_DIRTY RESULT_VARIABLE STATUS_RESULT)
if(NOT GIT_RESULT EQUAL 0 OR NOT STATUS_RESULT EQUAL 0 OR NOT ACTUAL_COMMIT STREQUAL JUCE_COMMIT OR NOT JUCE_DIRTY STREQUAL "")
    message(FATAL_ERROR "JUCE must be the clean checkout pinned in desktop.lock.json")
endif()
file(READ "${REPO_ROOT}/effects/catalog.json" CATALOG)
string(JSON COUNT LENGTH "${CATALOG}" effects)
math(EXPR LAST "${COUNT} - 1")
set(FOUND FALSE)
foreach(INDEX RANGE ${LAST})
    string(JSON ID GET "${CATALOG}" effects ${INDEX} id)
    if(ID STREQUAL EFFECT)
        set(FOUND TRUE)
        string(JSON SOURCE GET "${CATALOG}" effects ${INDEX} source)
        string(JSON PRODUCT_NAME GET "${CATALOG}" effects ${INDEX} desktop productName)
        string(JSON PLUGIN_CODE GET "${CATALOG}" effects ${INDEX} desktop pluginCode)
        foreach(PARAM RANGE 0 2)
            string(JSON PARAM${PARAM} GET "${CATALOG}" effects ${INDEX} desktop parameterNames ${PARAM})
        endforeach()
    endif()
endforeach()
if(NOT FOUND)
    message(FATAL_ERROR "Unknown catalog effect ${EFFECT}")
endif()
configure_file(EffectConfig.h.in "${CMAKE_CURRENT_BINARY_DIR}/generated/EffectConfig.h" @ONLY)
set(JUCE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(JUCE_BUILD_EXTRAS OFF CACHE BOOL "" FORCE)
add_subdirectory("${JUCE_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/juce")
juce_add_plugin(EndlessPlugin
    COMPANY_NAME "Ryan Veitch"
    BUNDLE_ID "com.rveitch.endless.${EFFECT}"
    PLUGIN_MANUFACTURER_CODE Rvch
    PLUGIN_CODE "${PLUGIN_CODE}"
    PRODUCT_NAME "${PRODUCT_NAME}"
    FORMATS VST3
    IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
    COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx)
target_sources(EndlessPlugin PRIVATE Processor.cpp "${REPO_ROOT}/${SOURCE}")
target_include_directories(EndlessPlugin PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(EndlessPlugin PUBLIC ENDLESS_DESKTOP=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_REPORT_APP_USAGE=0)
target_link_libraries(EndlessPlugin PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
# Keep our code strict without changing JUCE's warning policy.
set_source_files_properties(Processor.cpp "${REPO_ROOT}/${SOURCE}" PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
add_executable(ProcessorTest tests/ProcessorTest.cpp)
target_include_directories(ProcessorTest PRIVATE "${JUCE_SOURCE_DIR}/modules")
target_compile_definitions(ProcessorTest PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_compile_options(ProcessorTest PRIVATE -Wall -Wextra -Werror)
target_link_libraries(ProcessorTest PRIVATE EndlessPlugin)
enable_testing()
if(EFFECT STREQUAL "junoChorus")
    add_test(NAME ProcessorTest COMMAND ProcessorTest)
    set_tests_properties(ProcessorTest PROPERTIES TIMEOUT 30)
endif()
